from ..database import get_db
from .. import models, schemas
from fastapi import Depends, HTTPException, status, Response, APIRouter
from sqlalchemy.orm import Session
from typing import List

router = APIRouter(
    prefix= "/posts",
    tags=["Posts control"]
)

#Root

@router.get("/root")
def root():
    return "root"

#--------All 'Get' path operations----------

#Get all posts
@router.get("/", response_model=List[schemas.PostResponse])
def get_posts(db: Session = Depends(get_db)):

    posts = db.query(models.Post).all()

    return posts

#Get one post by id
@router.get("/{id}", response_model=schemas.PostResponse)
def get_post(id: int, db: Session = Depends(get_db)):

    post = db.query(models.Post).filter(models.Post.id == id).first()

    if not post:
        raise HTTPException(status.HTTP_404_NOT_FOUND, 
                            f"post with id: {id} does not exist")
    return post

#---------Post-----------
@router.post("/", status_code=status.HTTP_201_CREATED ,response_model=schemas.PostResponse)
def create_post(post: schemas.PostCreate, db: Session = Depends(get_db)):

    new_post = models.Post(**post.dict())

    db.add(new_post)
    db.commit()
    db.refresh(new_post)

    return new_post

#----------------Delete---------------------
@router.delete("/{id}")
def delete_post(id: int, db: Session = Depends(get_db)):

    post_query = db.query(models.Post).filter(models.Post.id == id)

    if not post_query.first():
        raise HTTPException(404, f"post with id: {id} does not exist")

    post_query.delete(synchronize_session=False)
    db.commit()

    return Response(status_code= status.HTTP_204_NO_CONTENT)

#--------------Update----------------
@router.put("/{id}", status_code=201, response_model=schemas.PostResponse)
def updated_post(updated_post: schemas.PostCreate, id: int, db: Session = Depends(get_db)):

    post_query = db.query(models.Post).filter(models.Post.id == id)
    post = post_query.first()

    if not post:
        raise HTTPException(status.HTTP_404_NOT_FOUND, f"post with id: {id} does not exist")

    post_query.update(updated_post.dict())

    return post_query.first()
