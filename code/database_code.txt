#create an engine that is responsible for managing database connection
from sqlalchemy import create_engine

sql_database_url = 'postgresql://postgres:ha12hu12@LocalHost/fastapi_hu'

engine = create_engine(sql_database_url)

#create a factory that creates sessions bound to the engine whenever i need

from sqlalchemy.orm import sessionmaker

SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine)

#make a 'base' variable so model classes (ORM classes that film the tables) 
# inherit from it

from sqlalchemy.ext.declarative import declarative_base

Base = declarative_base()

#make a function that opens a 
# session then give to FastAPI to do the work then close session
def get_db():
    db = SessionLocal()

    try:
        yield db
    finally:
        db.close()
